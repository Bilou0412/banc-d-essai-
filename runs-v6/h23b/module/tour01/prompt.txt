Les tests sont en lecture seule.

Contexte technique (identique pour toutes les sondes) :
- Python 3.13, bibliothèque standard + le paquet `cryptography` (déjà installé). Rien d'autre.
- Tout le système tient dans un seul module `app.py`. Pas de réseau réel ni de disque : les
  échanges entre machines passent par un réseau simulé en mémoire, et le temps est une valeur
  que les tests font avancer.
- Tests : pytest, dans `tests/`, qui importent `app`. Chaque test s'exécute en moins de 20 s.

Mécanisme à réaliser pour de vrai (interdit de le simuler) : la persistance entre deux ouvertures.
- Exception à « pas de disque » : l'application enregistre ses données dans un fichier dont le chemin
  est donné par les tests (par ex. `tmp_path`) ; une nouvelle instance créée sur ce chemin relit tout.
- Rien ne passe d'une instance à l'autre par la mémoire (pas de variable globale ni de cache partagé).

def _png():
    def chunk(t, d):
        return (struct.pack(">I", len(d)) + t + d
                + struct.pack(">I", zlib.crc32(t + d) & 0xFFFFFFFF))
    ihdr = struct.pack(">IIBBBBB", 1, 1, 8, 2, 0, 0, 0)
    return (b"\x89PNG\r\n\x1a\n" + chunk(b"IHDR", ihdr)
            + chunk(b"IDAT", zlib.compress(b"\x00\xff\x00\x00")) + chunk(b"IEND", b""))


PNG = _png()

def titles(inst):
    return sorted(r["title"] for r in inst.list_recipes())


def ing(inst, rid, servings=None):
    fiche = inst.recipe(rid) if servings is None else inst.recipe(rid, servings=servings)
    return {i["name"]: i for i in fiche["ingredients"]}


def qty(s):
    return float(s)


def shop(inst):

def find_id(inst, title):
    ids = [r["id"] for r in inst.list_recipes() if r["title"] == title]
    assert len(ids) == 1, f"attendu une recette '{title}', obtenu {inst.list_recipes()!r}"
    return ids[0]


def run_fresh_process(path, code):
    """Execute `code` dans un AUTRE processus Python : aucune memoire partagee possible."""
    app_dir = os.path.dirname(os.path.abspath(app.__file__))
    env = dict(os.environ)
    env["PYTHONPATH"] = app_dir + os.pathsep + env.get("PYTHONPATH", "")
    script = "import app, json, sys\na = app.App(sys.argv[1])\n" + code
    out = subprocess.run([sys.executable, "-c", script, str(path)], capture_output=True,
                         text=True, env=env, timeout=15)
    assert out.returncode == 0, f"processus neuf en echec : {out.stderr}"
    return json.loads(out.stdout.strip().splitlines()[-1])

def _png():
    def chunk(t, d):
        return (struct.pack(">I", len(d)) + t + d
                + struct.pack(">I", zlib.crc32(t + d) & 0xFFFFFFFF))
    ihdr = struct.pack(">IIBBBBB", 1, 1, 8, 2, 0, 0, 0)
    return (b"\x89PNG\r\n\x1a\n" + chunk(b"IHDR", ihdr)
            + chunk(b"IDAT", zlib.compress(b"\x00\xff\x00\x00")) + chunk(b"IEND", b""))

    app_dir = os.path.dirname(os.path.abspath(app.__file__))

    env["PYTHONPATH"] = app_dir + os.pathsep + env.get("PYTHONPATH", "")
    script = "import app, json, sys, hashlib\na = app.App(sys.argv[1])\n" + code

                         text=True, env=env, timeout=15)
