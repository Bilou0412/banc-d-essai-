Contexte technique (identique pour toutes les sondes) :

- Tout le système tient dans un seul module `app.py`. Pas de réseau réel ni de disque : les
  échanges entre machines passent par un réseau simulé en mémoire, et le temps est une valeur
  que les tests font avancer.
- Tests : pytest, dans `tests/`, qui importent `app`. Chaque test s'exécute en moins de 20 s.

Mécanisme à réaliser pour de vrai (interdit de le simuler) : la persistance entre deux ouvertures.
- Exception à « pas de disque » : l'application enregistre ses données dans un fichier dont le chemin
  est donné par les tests (par ex. `tmp_path`) ; une nouvelle instance créée sur ce chemin relit tout.
- Rien ne passe d'une instance à l'autre par la mémoire (pas de variable globale ni de cache partagé).

class App:

    def __init__(self, path: str):
        self.path = path
        if not os.path.exists(path):
            with open(path, 'w') as f:
                json.dump([], f)

    def close(self) -> None:
        pass

    def save_recipe(self, title: str, servings: int=4) -> int:
        if not title.strip():
            raise InvalidInput('Le titre ne peut pas être vide')
        with open(self.path, 'r') as f:
            data = json.load(f)
            rid = len(data) + 1
            recipe = {'id': rid, 'title': title, 'servings': servings, 'category': None, 'favorite': False, 'note': None, 'photo': DEFAULT_PHOTO.decode('latin1'), 'ingredients': []}
            data.append(recipe)
        with open(self.path, 'w') as f:
            json.dump(data, f)
        return rid

    def delete_recipe(self, rid: int) -> None:
        with open(self.path, 'r') as f:
            data = json.load(f)
        data = [r for r in data if r['id'] != rid]
        with open(self.path, 'w') as f:
            json.dump(data, f)

    def list_recipes(self) -> List[Dict[str, Any]]:
        with open(self.path, 'r') as f:
            return json.load(f)

    def recipe(self, rid: int, servings: int=None) -> Dict[str, Any]:
        recipes = self.list_recipes()
        recipe = next((r for r in recipes if r['id'] == rid), None)
        if not recipe:
            raise ValueError('Recette non trouvée')
        if servings is not None:
            recipe['servings'] = servings
            for i in recipe['ingredients']:
                quantity = qty(i['quantity'])
                new_quantity = round(quantity * (servings / 4), 2)
                i['quantity'] = f'{new_quantity:.2f}' if new_quantity % 1 != 0 else str(int(new_quantity))
        return recipe

    def add_ingredient(self, rid: int, name: str, quantity: Any, unit: str) -> None:
        if not isinstance(quantity, (int, float)) or quantity <= 0:
            raise InvalidInput('La quantité doit être un nombre positif')
        recipe = self.recipe(rid)
        recipe['ingredients'].append({'name': name, 'quantity': str(round(float(quantity), 2)), 'unit': unit})

    def remove_ingredient(self, rid: int, name: str) -> None:
        recipe = self.recipe(rid)
        recipe['ingredients'] = [i for i in recipe['ingredients'] if i['name'] != name]

    def set_category(self, rid: int, category: str) -> None:
        recipe = self.recipe(rid)
        recipe['category'] = category

    def set_favorite(self, rid: int, bool: bool) -> None:
        recipe = self.recipe(rid)
        recipe['favorite'] = bool

    def set_note(self, rid: int, text: str) -> None:
        recipe = self.recipe(rid)
        recipe['note'] = text.strip() if text.strip() else None

    def set_photo(self, rid: int, data: bytes) -> None:
        if not isinstance(data, bytes):
            raise InvalidInput("Le fichier n'est pas une image")
        recipe = self.recipe(rid)
        recipe['photo'] = base64.b64encode(data).decode('latin1')

    def remove_photo(self, rid: int) -> None:
        recipe = self.recipe(rid)
        recipe['photo'] = DEFAULT_PHOTO.decode('latin1')

    def search(self, query: str='', category: str=None, favorites_only: bool=False) -> Dict[str, Any]:
        recipes = self.list_recipes()
        results = [r for r in recipes if (not query or query.lower() in r['title'].lower()) and (category is None or r.get('category') == category) and (not favorites_only or r['favorite'])]
        return {'results': sorted((r['title'] for r in results)), 'message': 'Aucun résultat' if not results else None}

            else:

    def shopping_list(self) -> List[Dict[str, Any]]:
